cmake_minimum_required(VERSION 3.20)
project(dce VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/deps/boost_1_86_0/boost/asio.hpp" OR NOT EXISTS "${CMAKE_SOURCE_DIR}/deps/json.hpp")
  message(FATAL_ERROR "Run sh scripts/deps.sh first (pinned, SHA256-verified dependencies)")
endif()
add_library(dce src/protocol.cpp src/compute.cpp src/scheduler.cpp src/http.cpp)
target_include_directories(dce PUBLIC include)
target_include_directories(dce SYSTEM PUBLIC deps deps/boost_1_86_0)
target_compile_definitions(dce PUBLIC BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
target_link_libraries(dce PUBLIC Threads::Threads)
target_compile_options(dce PRIVATE -Wall -Wextra -Wpedantic)
foreach(app coordinator worker cli)
  add_executable(dce-${app} src/${app}.cpp)
  target_link_libraries(dce-${app} PRIVATE dce)
endforeach()
enable_testing()
add_executable(dce-tests tests/unit.cpp)
target_link_libraries(dce-tests PRIVATE dce)
add_test(NAME unit COMMAND dce-tests)
add_executable(dce-verify tests/integration.cpp)
target_link_libraries(dce-verify PRIVATE dce)
add_test(NAME integration COMMAND dce-verify $<TARGET_FILE_DIR:dce-worker> ${CMAKE_SOURCE_DIR})
set_tests_properties(integration PROPERTIES TIMEOUT 180)
