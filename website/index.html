Core Capabilities

What It Does

The engine covers the essential building blocks of a distributed job runner.

Master Node

Accepts job submissions over TCP, manages the worker registry, and routes work to available nodes.

FIFO Scheduling

When all workers are busy, jobs enter a FIFO queue and are dispatched as workers become idle.

Heartbeat Monitoring

Workers send periodic heartbeat messages so the master can detect unavailable nodes within seconds.

Failure Recovery

If a worker dies, the master requeues its running job and reassigns it to a healthy node automatically.

Docker Execution

Worker nodes execute Python jobs inside isolated Docker containers for sandboxed, reproducible runs.

CLI Shell

An interactive command shell (dcesh) for submitting jobs, checking status, and reading logs.

job_queue = []  # FIFO order of waiting jobs

def handle_message(msg: str) -> str:
    parts = msg.strip().split()
    cmd = parts[0]

    if cmd == "SUBMIT":
        job_id = parts[1]
        code = msg.split("\n", 1)[1]
        jobs[job_id] = {"code": code, "status": "QUEUED"}
        assigned = assign_job(job_id, code)
        if assigned:
            return f"JOB {job_id} ASSIGNED TO {assigned}"
        job_queue.append(job_id)
        return f"JOB {job_id} QUEUED"

if __name__ == "__main__":

    threading.Thread(target=worker_monitor, daemon=True).start()

import subprocess, tempfile

def run_job_in_docker(job_id, code):
    # Save code to temporary file
    with tempfile.NamedTemporaryFile(
        delete=False, suffix=".py"
    ) as tmp:
        tmp.write(code.encode())
        tmp_path = tmp.name

    # Execute inside Docker container
    cmd = [
        "docker", "run", "--rm",
        "-v", f"{tmp_path}:/app/job.py",
        "python:3.10",
        "python", "/app/job.py"
    ]

    output = subprocess.check_output(cmd, stderr=subprocess.STDOUT)
    return output.decode()

def handle_master_message(msg: str) -> str:
    parts = msg.split()
    if parts[0] == "RUN":
        job_id = parts[1]
        code = msg.split("\n", 1)[1]

protocol/tcp_server.py & tcp_client.py Python 
import socket, threading

class TCPServer:
    def __init__(self, host, port, handler):
        self.host = host
        self.port = port
        self.handler = handler

    def start(self):
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server.bind((self.host, self.port))
        server.listen(5)

        while True:
            client_socket, addr = server.accept()
            threading.Thread(
                target=self._handle_client,
                args=(client_socket, addr),
                daemon=True
            ).start()

class TCPClient:
    def send(self, data: str) -> str:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((self.host, self.port))
        sock.sendall(data.encode())
        # Read full response
        response = []
        while True:
            chunk = sock.recv(4096)
            if not chunk: break
            response.append(chunk.decode())
        sock.close()
        return "".join(response)
master/master.py — Scheduler Python 
def assign_job(job_id, code):
    """Assign a job to the first idle, alive worker."""
    for wid, w in workers.items():
        if w["status"] == "idle" and w["last_heartbeat"] != 0:
            client = TCPClient(w["host"], w["port"])
            payload = f"RUN {job_id} {len(code)}\n{code}"
            client.send(payload)
            workers[wid]["status"] = "busy"
            jobs[job_id]["status"] = "RUNNING"
            jobs[job_id]["worker"] = wid
            return wid
    return None

def scheduler_loop():
    """Continuously try to assign queued jobs."""
    while True:
        time.sleep(1)
        if not job_queue:
            continue
        job_id = job_queue[0]
        code = jobs[job_id]["code"]
        assigned = assign_job(job_id, code)
        if assigned:
            job_queue.pop(0)
master/master.py — Failure Detection Python 
HEARTBEAT_TIMEOUT = 15  # seconds

def worker_monitor():
    """Detect dead workers and requeue their jobs."""
    while True:
        time.sleep(5)
        now = time.time()

        for wid, w in workers.items():
            if w["last_heartbeat"] == 0:
                continue

            # Worker timed out → mark dead
            if now - w["last_heartbeat"] > HEARTBEAT_TIMEOUT:
                if w["status"] != "dead":
                    w["status"] = "dead"

                    # Requeue any job the dead worker had
                    for job_id, job in jobs.items():
                        if job["worker"] == wid \
                           and job["status"] == "RUNNING":
                            job["status"] = "QUEUED"
                            job["worker"] = None
                            job_queue.append(job_id)

Takeaways

What I Learned

01

Master/Worker Architecture

How to structure a central coordinator that manages worker state, routes tasks, and handles concurrent TCP connections using threads.

02

Heartbeat-Based Failure Detection

How periodic health signals enable a coordinator to detect silent failures and trigger recovery actions within a configurable timeout window.

03

Job Scheduling Under Constraints

How scheduling decisions change when workers can be idle, busy, or dead — and why a simple FIFO queue is a reasonable starting point.

04

Remote Execution & Result Collection

How to safely execute code on remote nodes, capture output, and surface results through a client interface.

05

Why Production Systems Need More

Why real distributed systems require stronger protocols, persistent storage, retries, idempotency, and observability beyond what this project covers.
